=== 01
- C のプログラムは `main` 関数から動きはじめます。
- `#include <stdio.h>` は、「表示などの道具を使います」という宣言です。`printf` を使うのに必要です。
- `printf` は自動では改行しません。改行したい所に `\n` を書きます。
- 文の終わりには `;` が必要です。`/* */` の間と、`//` から行の終わりまではコメントです。
- C は、コンパイル（機械の言葉に翻訳）してから動かします。`gcc main.c -o main` で翻訳、`./main` で実行です。
--- main
/* はじめてのプログラム */
#include <stdio.h>

int main(void) {
    printf("こんにちは！\n");
    printf("プログラミングをはじめます。\n");
    printf("1 + 2 = %d\n", 1 + 2);   // %d の所に、後ろの値が入ります
    return 0;                        // 0 は「問題なく終わった」のしるし
}

=== 02
- 変数を作るときは、`int age = 15;` のように、先に型を書きます。
- よく使う型: `int`（整数）、`double`（小数）、`char`（1文字）、`bool`（真・偽。`<stdbool.h>` が必要）。
- 文字列の専用の型はなく、`char` の配列で表します（`char name[] = "..."`）。
- `printf` では、型に合わせて `%d`（整数）、`%f`（小数）、`%s`（文字列）、`%c`（1文字）を使い分けます。
- 値を入れずに使った変数の中身は「でたらめ」です。必ず最初に値を入れましょう。
--- main
#include <stdio.h>
#include <stdbool.h>

int main(void) {
    char name[] = "さくら";      // 文字列（char の配列）
    int age = 15;                // 整数
    double height = 152.5;       // 小数
    bool is_student = true;      // 真か偽か（中身は 1 か 0）

    printf("名前: %s\n", name);
    printf("年れい: %d\n", age);
    printf("身長: %.1f\n", height);      // %.1f は「小数点より下を1けた」
    printf("学生: %d\n", is_student);    // true は 1 と表示されます

    age = age + 1;               // 変数は、あとから入れなおせます
    printf("来年は %d さい\n", age);
    return 0;
}

=== 03
- 整数どうしの `/` は、小数点より下を切りすてます（7 / 2 は 3）。小数がほしいときは、どちらかを小数にします。
- `%` は割り算のあまりです。表示で `%` そのものを出したいときは `%%` と書きます。
- 文字列は `+` でつなげられません。`snprintf` で、新しい入れ物に書きこんで作ります。
- `strlen` は文字列の長さ（バイト数）を返します。日本語は1文字が3バイトなので、文字数とは合いません。
--- main
#include <stdio.h>
#include <string.h>

int main(void) {
    int a = 7;
    int b = 2;
    printf("7 + 2 = %d\n", a + b);
    printf("7 - 2 = %d\n", a - b);
    printf("7 * 2 = %d\n", a * b);
    printf("7 / 2 = %d\n", a / b);            // 整数どうしは切りすて
    printf("7 / 2.0 = %.1f\n", a / 2.0);      // 小数にすると 3.5
    printf("7 %% 2 = %d\n", a % b);           // あまり

    char family[] = "山田";
    char given[] = "太郎";
    char full[64];                            // つないだ結果を入れる場所
    snprintf(full, sizeof(full), "%s %s", family, given);
    printf("フルネーム: %s\n", full);
    printf("文字数: %zu\n", strlen("program"));
    return 0;
}

=== 04
- `if (条件) { }` の形です。`else if` で次の条件、`else` で「どれでもないとき」です。
- C では、0 が「偽」、0 以外が「真」です。
- 等しいかどうかは `==`。`if (x = 5)` と `=` 1つで書くと「入れる」になり、いつも真になってしまいます。
- 「かつ」は `&&`、「または」は `||`、「〜でない」は `!` です。
- 文字列が同じかどうかは `==` ではなく `strcmp(a, b) == 0` で調べます。
--- main
#include <stdio.h>
#include <stdbool.h>
#include <string.h>

int main(void) {
    int score = 72;

    if (score >= 80) {
        printf("よくできました\n");
    } else if (score >= 60) {
        printf("合格\n");
    } else {
        printf("もう少し\n");
    }

    int age = 15;
    bool has_ticket = true;
    if (age >= 12 && has_ticket) {
        printf("入場できます\n");
    }

    char answer[] = "yes";
    if (strcmp(answer, "yes") == 0) {     // 文字列は strcmp でくらべる
        printf("ようこそ\n");
    }
    return 0;
}

=== 05
- `for (はじめ; 続ける条件; 毎回すること)` の3つを `;` で区切ります。
- `i++` は「i を 1 ふやす」、`count--` は「1 へらす」です。
- `while (条件)` は、条件が成り立つあいだくりかえします。
- `continue` は「今回は飛ばして次へ」、`break` は「くりかえしをやめる」です。
--- main
#include <stdio.h>

int main(void) {
    // 決まった回数くりかえす
    for (int i = 1; i <= 5; i++) {
        printf("%d 回目\n", i);
    }

    // 条件が成り立つあいだくりかえす
    int count = 3;
    while (count > 0) {
        printf("%d\n", count);
        count--;
    }
    printf("発射！\n");

    // 1〜10 の偶数だけを足す
    int total = 0;
    for (int i = 1; i <= 10; i++) {
        if (i % 2 == 1) {
            continue;      // 奇数は飛ばす
        }
        total += i;
    }
    printf("偶数の合計: %d\n", total);
    return 0;
}

=== 06
- `int add(int a, int b)` は、「int を2つ受けとって、int を返す」関数です。
- 何も返さない関数は `void`、引数がない関数は `(void)` と書きます。
- 関数は、使う所より前に書きます。あとに書きたいときは、先に「プロトタイプ宣言」（形だけ）を書いておきます。
- 引数には、値の「コピー」が渡されます。関数の中で変えても、呼んだ側の変数は変わりません。
--- main
#include <stdio.h>
#include <stdbool.h>

// プロトタイプ宣言: 「あとでこういう関数が出てきます」と先に知らせる
int add(int a, int b);
bool is_even(int n);
void greet(const char *name);

int main(void) {
    printf("3 + 4 = %d\n", add(3, 4));
    greet("さくら");
    printf("10 は偶数？ %d\n", is_even(10));   // 1 が真、0 が偽
    printf("7 は偶数？ %d\n", is_even(7));
    return 0;
}

int add(int a, int b) {
    return a + b;
}

bool is_even(int n) {
    return n % 2 == 0;
}

void greet(const char *name) {          // 何も返さない
    printf("こんにちは、%sさん\n", name);
}

=== 07
- 配列は `int scores[10];` のように、型と大きさを決めて作ります。あとから大きさは変えられません。
- 番号は 0 から始まります。大きさ 10 の配列で使えるのは `[0]` から `[9]` までです。
- C は「はんいの外」をチェックしません。`scores[10]` に書くと、ほかのデータをこわしてしまいます。いちばん気をつける所です。
- 配列は自分の長さをおぼえていません。「いま何個入っているか」は、自分で変数に持ちます。
--- main
#include <stdio.h>

#define MAX 10     // 入れ物の大きさ

int main(void) {
    int scores[MAX] = {70, 85, 60};
    int count = 3;                 // いま入っている数（自分で数える）

    if (count < MAX) {             // 空きがあるか確かめてから追加
        scores[count] = 90;
        count++;
    }

    printf("最初の点数: %d\n", scores[0]);
    printf("人数: %d\n", count);

    int total = 0;
    int max = scores[0];
    for (int i = 0; i < count; i++) {
        total += scores[i];
        if (scores[i] > max) {
            max = scores[i];
        }
    }
    printf("合計: %d\n", total);
    printf("平均: %.2f\n", (double)total / count);
    printf("最高点: %d\n", max);
    return 0;
}

=== 08
- C には、辞書（マップ）が最初から入っていません。「キーと値を持つ構造体」の配列を作り、自分で探します。
- `struct` は、いくつかのデータを1つにまとめた型です。`.` で中身を取り出します。
- 文字列のキーは `strcmp` でくらべます。
- 「見つからなかった」ことは、ありえない値（ここでは -1）を返して知らせています。
- ほかの言語の辞書が中で何をしているかが、C で書くとよくわかります。
--- main
#include <stdio.h>
#include <string.h>

struct Item {
    char name[32];
    int price;
};

// 名前で探して、値段を返す。なければ -1
int find_price(const struct Item items[], int count, const char *name) {
    for (int i = 0; i < count; i++) {
        if (strcmp(items[i].name, name) == 0) {
            return items[i].price;
        }
    }
    return -1;
}

int main(void) {
    struct Item items[10] = {{"りんご", 120}, {"バナナ", 80}};
    int count = 2;

    strcpy(items[count].name, "みかん");    // 追加
    items[count].price = 60;
    count++;

    printf("りんご は %d 円\n", find_price(items, count, "りんご"));

    if (find_price(items, count, "ぶどう") == -1) {
        printf("ぶどう はありません\n");
    }

    printf("--- 一覧 ---\n");
    for (int i = 0; i < count; i++) {
        printf("%s: %d 円\n", items[i].name, items[i].price);
    }
    printf("種類: %d\n", count);
    return 0;
}

=== 09
- C の文字列は「`char` の配列」で、最後に終わりのしるし `\0` が入っています。
- 文字列の道具は `<string.h>` にあります。`strlen`（長さ）、`strcpy`（コピー）、`strcmp`（くらべる）、`strstr`（探す）など。
- `strtok` は、文字列を区切りで分けます。もとの文字列を書きかえるので、コピーしてから使います。
- 入れ物の大きさをこえて書きこむと、プログラムがこわれます。`snprintf` のように、大きさを渡せる関数を使いましょう。
--- main
#include <stdio.h>
#include <string.h>
#include <ctype.h>

int main(void) {
    char text[] = "apple,banana,cherry";

    // 分ける（strtok はもとの文字列を書きかえるので、コピーを使う）
    char copy[64];
    snprintf(copy, sizeof(copy), "%s", text);
    char *fruits[10];
    int count = 0;
    for (char *p = strtok(copy, ","); p != NULL; p = strtok(NULL, ",")) {
        fruits[count] = p;
        count++;
    }
    printf("%d 個\n", count);
    printf("1番目: %s\n", fruits[0]);

    // 大文字にする（1文字ずつ変える）
    char upper[32];
    size_t len = strlen(fruits[0]);
    for (size_t i = 0; i < len; i++) {
        upper[i] = (char)toupper((unsigned char)fruits[0][i]);
    }
    upper[len] = '\0';                  // 終わりのしるしを忘れずに
    printf("大文字: %s\n", upper);

    // ふくまれているか
    printf("banana をふくむ？ %d\n", strstr(text, "banana") != NULL);

    // つなぐ
    char joined[64] = "";
    for (int i = 0; i < count; i++) {
        if (i > 0) {
            strncat(joined, " / ", sizeof(joined) - strlen(joined) - 1);
        }
        strncat(joined, fruits[i], sizeof(joined) - strlen(joined) - 1);
    }
    printf("つなぐ: %s\n", joined);
    return 0;
}

=== 10
- C にはクラスがありません。かわりに、データを `struct` にまとめ、それをあつかう関数を作ります。
- `typedef struct { ... } Account;` と書くと、`Account` という名前の型として使えます。
- 関数に構造体を渡すと「コピー」が渡されます。中身を変えたいときは、`&a` で「場所（アドレス）」を渡します。
- ポインタから中身を取り出すときは `->` を使います（`acc->balance`）。
--- main
#include <stdio.h>

typedef struct {
    char owner[32];
    int balance;
} Account;

// 中身を変えるので、ポインタ（場所）で受けとる
void deposit(Account *acc, int amount) {
    acc->balance += amount;
}

// 読むだけなので const をつける
void show(const Account *acc) {
    printf("%s の残高: %d 円\n", acc->owner, acc->balance);
}

int main(void) {
    Account a = {"さくら", 1000};
    Account b = {"たろう", 300};

    deposit(&a, 500);      // a の場所を渡す。a だけが増える
    show(&a);
    show(&b);
    return 0;
}

=== 11
title: ポインタ
goal: 「変数の場所（アドレス）」をあつかうポインタを使って、関数から変数を書きかえられるようになる。
intro: ポインタは C でいちばん大事で、いちばんつまずきやすい所です。ポインタとは「変数がメモリのどこにあるか」という場所の情報です。場所がわかれば、はなれた所からその変数を読んだり書いたりできます。配列も文字列も、関数への受け渡しも、C ではすべてポインタで動いています。
quiz: int x = 5; int *p = &x; *p = 9; のあと、x の値は？ ;; 5 ;; 9 ;; 場所を表す数 ;; 2 ;; p は x の場所を指しています。*p = 9 は「p が指している場所に 9 を入れる」なので、x が 9 になります。
task: int の配列と個数を受けとり、全部の要素を2倍にする関数 double_all を作ってみよう。
hint: void double_all(int *nums, int count) の中で、nums[i] *= 2; とします。配列を渡すと、先頭の場所が渡されます。
- `&x` は「x の場所（アドレス）」、`*p` は「p が指している場所の中身」です。
- `int *p` は「int の場所を入れる変数」。これがポインタです。
- 関数に場所を渡すと、関数の中から呼んだ側の変数を書きかえられます（`swap` が定番の例）。
- 配列の名前は、先頭の要素の場所を表します。`nums[i]` と `*(nums + i)` は同じ意味です。
- どこも指していないポインタには `NULL` を入れます。`NULL` や、もう無い場所を `*` で読むと、プログラムが落ちます。
--- main
#include <stdio.h>

// 2つの変数の中身を入れかえる（場所を受けとる）
void swap(int *a, int *b) {
    int tmp = *a;
    *a = *b;
    *b = tmp;
}

// 配列は「先頭の場所」として渡される
int sum(const int *nums, int count) {
    int total = 0;
    for (int i = 0; i < count; i++) {
        total += nums[i];
    }
    return total;
}

int main(void) {
    int x = 5;
    int *p = &x;            // p は x の場所を指す

    printf("x = %d\n", x);
    printf("*p = %d\n", *p);    // p が指す場所の中身

    *p = 9;                 // p を通して x を書きかえる
    printf("x = %d\n", x);

    int a = 1, b = 2;
    swap(&a, &b);           // 場所を渡す
    printf("a = %d, b = %d\n", a, b);

    int nums[] = {10, 20, 30};
    printf("合計: %d\n", sum(nums, 3));
    printf("2番目: %d\n", *(nums + 1));   // nums[1] と同じ

    int *nothing = NULL;    // どこも指していない
    if (nothing == NULL) {
        printf("何も指していません\n");
    }
    return 0;
}

=== 12
- C では、関数の「形」をヘッダファイル（`.h`）に、「中身」をソースファイル（`.c`）に書いて分けます。
- 使う側は `#include "greeting.h"` と書きます。自分で作ったヘッダは `" "`、最初からある物は `< >` で囲みます。
- `#ifndef` 〜 `#endif` は、同じヘッダを2回読みこまないためのおまじないです（インクルードガード）。
- コンパイルするときは、`.c` ファイルを全部ならべます: `gcc main.c greeting.c -o main`
--- greeting.h
// あいさつを作る部品（関数の形だけを書く）
#ifndef GREETING_H
#define GREETING_H

void hello(const char *name);
void bye(const char *name);

#endif
--- greeting.c
// あいさつを作る部品（関数の中身を書く）
#include <stdio.h>
#include "greeting.h"

void hello(const char *name) {
    printf("こんにちは、%sさん\n", name);
}

void bye(const char *name) {
    printf("さようなら、%sさん\n", name);
}
--- main
#include "greeting.h"

int main(void) {
    hello("さくら");
    bye("さくら");
    return 0;
}

=== 13
- C には例外（try / catch）がありません。関数が「うまくいったか」を返す値で知らせ、呼ぶ側が必ず確かめます。
- よくある形: 成功なら 0、失敗なら 0 以外を返す。結果そのものは、ポインタの引数に書きこみます。
- 返す値を確かめるのを忘れると、失敗に気づかないまま動きつづけます。確かめるのは呼ぶ側の責任です。
- 文字列を数に変えるときは `strtol` を使うと、「数に変えられなかった」ことがわかります（`atoi` ではわかりません）。
--- main
#include <stdio.h>
#include <stdlib.h>

// 成功なら 0、失敗なら -1 を返す。答えは *result に入れる
int divide(int a, int b, int *result) {
    if (b == 0) {
        return -1;
    }
    *result = a / b;
    return 0;
}

// 文字列を数にする。成功なら 0、失敗なら -1
int parse_int(const char *text, int *result) {
    char *end;
    long value = strtol(text, &end, 10);
    if (end == text || *end != '\0') {    // 数として読めなかった
        return -1;
    }
    *result = (int)value;
    return 0;
}

int main(void) {
    int answer;

    if (divide(10, 2, &answer) == 0) {
        printf("10 ÷ 2 = %d\n", answer);
    }
    if (divide(10, 0, &answer) != 0) {
        printf("エラー: 0 で割ることはできません\n");
    }

    const char *text = "abc";
    int number;
    if (parse_int(text, &number) != 0) {
        printf("\"%s\" は数ではありません\n", text);
    }
    printf("チェック終わり\n");
    return 0;
}

=== 14
- ファイルは `fopen` で開き、使い終わったら必ず `fclose` で閉じます。
- `"w"` は書きこみ（中身は消えて新しくなる）、`"r"` は読みこみ、`"a"` は後ろに追加です。
- `fopen` は、開けなかったとき `NULL` を返します。必ず確かめてから使います。
- `fprintf` でファイルに書き、`fgets` で1行ずつ読みます。`fgets` は改行もいっしょに読みこみます。
--- main
#include <stdio.h>
#include <string.h>

int main(void) {
    const char *lines[] = {"牛乳を買う", "宿題をする", "本を読む"};
    int count = 3;

    // 書きこむ
    FILE *out = fopen("memo.txt", "w");
    if (out == NULL) {
        printf("memo.txt を作れませんでした\n");
        return 1;
    }
    for (int i = 0; i < count; i++) {
        fprintf(out, "%s\n", lines[i]);
    }
    fclose(out);                              // 必ず閉じる
    printf("%d 行 書きました\n", count);

    // 読みこむ
    FILE *in = fopen("memo.txt", "r");
    if (in == NULL) {
        return 1;
    }
    char buf[256];
    int no = 1;
    while (fgets(buf, sizeof(buf), in) != NULL) {
        buf[strcspn(buf, "\n")] = '\0';       // 行の終わりの改行を取る
        printf("%d: %s\n", no, buf);
        no++;
    }
    fclose(in);

    // ないファイルを開こうとしたとき
    FILE *none = fopen("ない.txt", "r");
    if (none == NULL) {
        printf("ない.txt は開けませんでした\n");
    }
    return 0;
}

=== 15
- `assert(条件)` は、条件が成り立たなければ、その場でプログラムを止めて、ファイル名と行番号を表示します。
- C には決まったテストの道具がないので、まずは `assert` で「こうなるはず」を書くのが手軽です。
- 仕事では Unity や CTest、GoogleTest などのテストの道具が使われます。
- テストしやすいように、「計算する関数」と「表示する所」を分けておきます。
--- main
#include <assert.h>
#include <stdio.h>
#include <string.h>

// n に合う言葉を out に書きこむ
void fizzbuzz(int n, char *out, size_t size) {
    if (n % 15 == 0) {
        snprintf(out, size, "FizzBuzz");
    } else if (n % 3 == 0) {
        snprintf(out, size, "Fizz");
    } else if (n % 5 == 0) {
        snprintf(out, size, "Buzz");
    } else {
        snprintf(out, size, "%d", n);
    }
}

// 「n のとき expected になるはず」を確かめる
void check(int n, const char *expected) {
    char actual[16];
    fizzbuzz(n, actual, sizeof(actual));
    assert(strcmp(actual, expected) == 0);
    printf("OK   %d は %s\n", n, expected);
}

int main(void) {
    check(3, "Fizz");
    check(5, "Buzz");
    check(15, "FizzBuzz");
    check(7, "7");
    printf("すべてのテストに合格\n");
    return 0;
}

=== 16
title: メモリを自分で管理する（malloc と free）
goal: 必要な大きさのメモリを malloc で借りて、free で返せるようになる。「あとから大きくなる配列」を作れるようになる。
intro: ふつうの配列は、大きさを先に決めなければなりません。「何個になるか動かしてみないとわからない」ときは、動いている最中にメモリを借ります。借りたメモリは、自分で返さなければなりません。返し忘れ（メモリリーク）や、返したあとに使ってしまうまちがいは、C の仕事でいちばん多い不具合の原因です。
quiz: malloc で借りたメモリを free しないまま、プログラムが長く動きつづけるとどうなる？ ;; 自動で返される ;; 使えるメモリが少しずつへっていく（メモリリーク） ;; すぐにコンパイルエラーになる ;; 2 ;; C には自動で片づけるしくみがありません。返し忘れたメモリは、プログラムが終わるまで使われたままになります。
task: 数を n 個入れられる配列を malloc で作り、1 から n までを入れて合計を表示し、最後に free する関数を書いてみよう。
hint: int *nums = malloc(n * sizeof(int)); のあと、NULL でないか確かめてから使います。
- `malloc(バイト数)` でメモリを借ります。失敗すると `NULL` が返るので、必ず確かめます。
- `sizeof(int)` は int 1つ分の大きさ。`n * sizeof(int)` で n 個分を借りられます。
- `realloc` は、借りたメモリの大きさを変えます。
- 使い終わったら `free` で返します。「1回借りたら、1回返す」がきまりです。
- 返したあとのポインタには `NULL` を入れておくと、まちがって使うのを防げます。
--- main
#include <stdio.h>
#include <stdlib.h>

// あとから大きくなる配列
typedef struct {
    int *data;       // 借りたメモリの場所
    int count;       // いま入っている数
    int capacity;    // 入れられる数
} IntList;

// 後ろに追加する。いっぱいなら、入れ物を2倍に広げる
int push(IntList *list, int value) {
    if (list->count == list->capacity) {
        int new_cap = list->capacity == 0 ? 2 : list->capacity * 2;
        int *bigger = realloc(list->data, (size_t)new_cap * sizeof(int));
        if (bigger == NULL) {
            return -1;           // 借りられなかった
        }
        list->data = bigger;
        list->capacity = new_cap;
        printf("  （入れ物を %d 個分に広げました）\n", new_cap);
    }
    list->data[list->count] = value;
    list->count++;
    return 0;
}

int main(void) {
    IntList list = {NULL, 0, 0};

    for (int i = 1; i <= 5; i++) {
        if (push(&list, i * 10) != 0) {
            printf("メモリが足りません\n");
            return 1;
        }
    }

    for (int i = 0; i < list.count; i++) {
        printf("%d ", list.data[i]);
    }
    printf("\n個数: %d\n", list.count);

    free(list.data);         // 借りたメモリを返す
    list.data = NULL;        // もう使えないしるし
    return 0;
}

=== 18
- 「単語に分ける → 数える → ならべる → 表示する」と、仕事ごとに関数を分けています。
- 辞書がないので、「単語と回数」の構造体の配列を使い、同じ単語を自分で探しています。
- ならべかえは `qsort`。「どちらを先にするか」を決める関数を渡します（関数ポインタ）。
- 入れ物の大きさ（`MAX_WORDS`、`MAX_LEN`）をこえないように、毎回確かめています。
--- main
#include <ctype.h>
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

#define MAX_WORDS 100
#define MAX_LEN 32

typedef struct {
    char word[MAX_LEN];
    int count;
} Entry;

// 単語を1つ数える。全体の種類の数を返す
int add_word(Entry entries[], int kinds, const char *word) {
    for (int i = 0; i < kinds; i++) {
        if (strcmp(entries[i].word, word) == 0) {
            entries[i].count++;
            return kinds;
        }
    }
    if (kinds < MAX_WORDS) {
        snprintf(entries[kinds].word, MAX_LEN, "%s", word);
        entries[kinds].count = 1;
        kinds++;
    }
    return kinds;
}

// 文章を単語に分けて数える。単語の総数を返す
int count_words(const char *text, Entry entries[], int *kinds) {
    char word[MAX_LEN];
    int len = 0, total = 0;
    for (const char *p = text;; p++) {
        if (isalpha((unsigned char)*p) && len < MAX_LEN - 1) {
            word[len++] = (char)tolower((unsigned char)*p);
        } else if (len > 0) {           // 単語の終わり
            word[len] = '\0';
            *kinds = add_word(entries, *kinds, word);
            total++;
            len = 0;
        }
        if (*p == '\0') {
            break;
        }
    }
    return total;
}

// 回数の多い順（同じならアルファベット順）
int compare(const void *a, const void *b) {
    const Entry *x = a, *y = b;
    if (x->count != y->count) {
        return y->count - x->count;
    }
    return strcmp(x->word, y->word);
}

// 成功なら 0、単語がなければ -1
int report(const char *text) {
    Entry entries[MAX_WORDS];
    int kinds = 0;
    int total = count_words(text, entries, &kinds);
    if (total == 0) {
        return -1;
    }
    qsort(entries, (size_t)kinds, sizeof(Entry), compare);
    printf("全部で %d 語、%d 種類\n", total, kinds);
    for (int i = 0; i < 3 && i < kinds; i++) {
        printf("%s: %d\n", entries[i].word, entries[i].count);
    }
    return 0;
}

int main(void) {
    report("The cat saw the dog. The dog saw the bird!");

    if (report("...") != 0) {
        printf("エラー: 単語がありません\n");
    }
    return 0;
}
